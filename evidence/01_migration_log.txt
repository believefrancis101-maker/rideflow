CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    role TEXT NOT NULL DEFAULT 'customer' CHECK (role IN ('customer', 'admin')),
    status TEXT NOT NULL DEFAULT 'active' CHECK (status IN ('active', 'disabled')),
    full_name TEXT NOT NULL CHECK (char_length(btrim(full_name)) > 0),
    email TEXT NOT NULL UNIQUE CHECK (email = lower(email) AND char_length(email) BETWEEN 3 AND 320),
    phone TEXT NULL CHECK (phone IS NULL OR char_length(phone) BETWEEN 7 AND 32),
    deleted_at TIMESTAMPTZ NULL CHECK (deleted_at IS NULL OR (status = 'disabled' AND deleted_at >= created_at)),
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE TABLE
CREATE TABLE driver_profiles (
    user_id UUID PRIMARY KEY REFERENCES users(id) ON DELETE RESTRICT,
    status TEXT NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'approved', 'suspended', 'rejected')),
    approved_at TIMESTAMPTZ NULL CHECK (status <> 'approved' OR approved_at IS NOT NULL) CHECK (approved_at IS NULL OR approved_at >= created_at),
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE TABLE
CREATE TABLE vehicles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    driver_id UUID NOT NULL REFERENCES driver_profiles(user_id) ON DELETE RESTRICT,
    license_plate TEXT NOT NULL UNIQUE CHECK (license_plate = upper(license_plate) AND char_length(license_plate) BETWEEN 2 AND 16),
    make TEXT NOT NULL CHECK (char_length(btrim(make)) > 0),
    model TEXT NOT NULL CHECK (char_length(btrim(model)) > 0),
    color TEXT NULL CHECK (color IS NULL OR char_length(color) <= 40),
    status TEXT NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'approved', 'suspended')),
    approved_at TIMESTAMPTZ NULL CHECK (status <> 'approved' OR approved_at IS NOT NULL) CHECK (approved_at IS NULL OR approved_at >= created_at),
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT vehicles_driver_id_id_unique UNIQUE (driver_id, id)

CREATE TABLE trips (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    passenger_id UUID NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
    driver_id UUID NULL REFERENCES driver_profiles(user_id) ON DELETE RESTRICT,
    vehicle_id UUID NULL REFERENCES vehicles(id) ON DELETE RESTRICT,
    status TEXT NOT NULL DEFAULT 'requested' CHECK (
        status IN ('requested', 'accepted', 'driver_arriving', 'in_progress', 'completed', 'cancelled', 'expired')
    ),
    pickup_latitude NUMERIC(9,6) NOT NULL CHECK (pickup_latitude BETWEEN -90 AND 90),
    pickup_longitude NUMERIC(9,6) NOT NULL CHECK (pickup_longitude BETWEEN -180 AND 180),
    dropoff_latitude NUMERIC(9,6) NOT NULL CHECK (dropoff_latitude BETWEEN -90 AND 90),
    dropoff_longitude NUMERIC(9,6) NOT NULL CHECK (dropoff_longitude BETWEEN -180 AND 180),
    pickup_address TEXT NULL CHECK (pickup_address IS NULL OR char_length(pickup_address) <= 500),
    dropoff_address TEXT NULL CHECK (dropoff_address IS NULL OR char_length(dropoff_address) <= 500),
    fare_minor BIGINT NOT NULL CHECK (fare_minor > 0),
    currency VARCHAR(3) NOT NULL CHECK (currency ~ '^[A-Z]{3}$'),
    payment_method TEXT NOT NULL CHECK (payment_method IN ('cash', 'electronic')),
    requested_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    accepted_at TIMESTAMPTZ NULL,
    driver_arriving_at TIMESTAMPTZ NULL,
    started_at TIMESTAMPTZ NULL,
    completed_at TIMESTAMPTZ NULL,
    cancelled_at TIMESTAMPTZ NULL,
    cancellation_reason TEXT NULL CHECK (cancellation_reason IS NULL OR char_length(cancellation_reason) <= 500),
    expired_at TIMESTAMPTZ NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    -- Composite foreign key enforcing that vehicle belongs to assigned driver
    CONSTRAINT trips_driver_vehicle_fkey FOREIGN KEY (driver_id, vehicle_id)
        REFERENCES vehicles(driver_id, id) ON DELETE RESTRICT,
    -- Driver and vehicle assignment pairing: both null or both set
    CONSTRAINT trips_driver_vehicle_paired CHECK (
        (driver_id IS NULL AND vehicle_id IS NULL) OR
        (driver_id IS NOT NULL AND vehicle_id IS NOT NULL)
    ),
    -- Passenger and driver must be distinct users
    CONSTRAINT trips_passenger_differs_from_driver CHECK (
        driver_id IS NULL OR passenger_id <> driver_id
    ),
    -- Status and timestamp coherence checks
    CONSTRAINT trips_requested_state_clean CHECK (
        status <> 'requested' OR (
            driver_id IS NULL AND accepted_at IS NULL AND driver_arriving_at IS NULL
            AND started_at IS NULL AND completed_at IS NULL AND cancelled_at IS NULL AND expired_at IS NULL
        )
    ),
    CONSTRAINT trips_assigned_state_requirements CHECK (
        status NOT IN ('accepted', 'driver_arriving', 'in_progress', 'completed') OR (
            driver_id IS NOT NULL AND accepted_at IS NOT NULL
        )
    ),
    CONSTRAINT trips_driver_arriving_requirements CHECK (
        status <> 'driver_arriving' OR driver_arriving_at IS NOT NULL
    ),
    CONSTRAINT trips_in_progress_requirements CHECK (
        status NOT IN ('in_progress', 'completed') OR started_at IS NOT NULL
    ),
    CONSTRAINT trips_completed_requirements CHECK (
        status <> 'completed' OR completed_at IS NOT NULL
    ),
    CONSTRAINT trips_cancelled_requirements CHECK (
        status <> 'cancelled' OR cancelled_at IS NOT NULL
    ),
    CONSTRAINT trips_expired_requirements CHECK (
        status <> 'expired' OR (expired_at IS NOT NULL AND driver_id IS NULL AND accepted_at IS NULL)
    ),
    CONSTRAINT trips_cancellation_reason_guard CHECK (
        status = 'cancelled' OR cancellation_reason IS NULL
    ),
    CONSTRAINT trips_terminal_mutually_exclusive CHECK (
        (completed_at IS NULL OR cancelled_at IS NULL) AND
        (completed_at IS NULL OR expired_at IS NULL) AND
        (cancelled_at IS NULL OR expired_at IS NULL)
    ),
    CONSTRAINT trips_chronological_accepted CHECK (
        accepted_at IS NULL OR accepted_at >= requested_at
    ),
    CONSTRAINT trips_chronological_arriving CHECK (
        driver_arriving_at IS NULL OR driver_arriving_at >= accepted_at
    ),
    CONSTRAINT trips_chronological_started CHECK (
        started_at IS NULL OR started_at >= accepted_at
    ),
    CONSTRAINT trips_chronological_completed CHECK (
        completed_at IS NULL OR completed_at >= started_at
    ),
    CONSTRAINT trips_chronological_cancelled CHECK (
        cancelled_at IS NULL OR cancelled_at >= requested_at
    ),
    -- Unique constraints supporting child composite foreign keys
    CONSTRAINT trips_id_passenger_id_unique UNIQUE (id, passenger_id),
    CONSTRAINT trips_id_fare_currency_method_unique UNIQUE (id, fare_minor, currency, payment_method)
);
CREATE TABLE

    longitude NUMERIC(9,6) NOT NULL CHECK (longitude BETWEEN -180 AND 180),
    accuracy_meters NUMERIC(7,2) NULL CHECK (accuracy_meters IS NULL OR accuracy_meters >= 0),
    recorded_at TIMESTAMPTZ NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE payments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    trip_id UUID NOT NULL REFERENCES trips(id) ON DELETE RESTRICT,
    attempt_no INTEGER NOT NULL CHECK (attempt_no > 0),
    method TEXT NOT NULL CHECK (method IN ('cash', 'electronic')),
    amount_minor BIGINT NOT NULL CHECK (amount_minor > 0),
    currency VARCHAR(3) NOT NULL CHECK (currency ~ '^[A-Z]{3}$'),
    status TEXT NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'succeeded', 'failed')),
    provider_reference TEXT NULL,
    failure_code TEXT NULL,
    settled_at TIMESTAMPTZ NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    -- Composite FK enforcing exact fare, currency, and payment method parity with trip
    CONSTRAINT payments_trip_fare_snapshot_fkey FOREIGN KEY (trip_id, amount_minor, currency, method)
        REFERENCES trips(id, fare_minor, currency, payment_method) ON DELETE RESTRICT,
    -- Sequence uniqueness per trip
    CONSTRAINT payments_trip_attempt_unique UNIQUE (trip_id, attempt_no),
    -- Settlement rules
    CONSTRAINT payments_settled_at_coherence CHECK (
        (status = 'pending' AND settled_at IS NULL) OR
        (status IN ('succeeded', 'failed') AND settled_at IS NOT NULL)
    ),
    -- Cash payment cannot have provider reference
    CONSTRAINT payments_cash_no_provider_ref CHECK (
        method <> 'cash' OR provider_reference IS NULL
    )
);
CREATE TABLE
CREATE TABLE reviews (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    trip_id UUID NOT NULL UNIQUE REFERENCES trips(id) ON DELETE RESTRICT,
    passenger_id UUID NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
    rating SMALLINT NOT NULL CHECK (rating BETWEEN 1 AND 5),
    comment TEXT NULL CHECK (comment IS NULL OR char_length(comment) <= 2000),
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    -- Composite FK enforcing that review author is strictly the passenger who took the trip
    CONSTRAINT reviews_trip_passenger_fkey FOREIGN KEY (trip_id, passenger_id)
        REFERENCES trips(id, passenger_id) ON DELETE RESTRICT
);
CREATE TABLE
CREATE TABLE audit_events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    actor_type TEXT NOT NULL CHECK (actor_type IN ('customer', 'admin', 'system')),
    actor_user_id UUID NULL REFERENCES users(id) ON DELETE RESTRICT,
    action TEXT NOT NULL CHECK (char_length(action) BETWEEN 1 AND 100),
    entity_type TEXT NOT NULL CHECK (entity_type ~ '^[a-z][a-z0-9_]{0,63}$'),
    entity_id UUID NOT NULL,
    before_data JSONB NULL,
    after_data JSONB NULL,
    metadata JSONB NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    -- System actor has no user ID; customer/admin actor requires user ID
    CONSTRAINT audit_events_actor_coherence CHECK (
        (actor_type = 'system' AND actor_user_id IS NULL) OR
        (actor_type IN ('customer', 'admin') AND actor_user_id IS NOT NULL)
    )
);
CREATE TABLE
CREATE TABLE idempotency_keys (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    principal_user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    operation TEXT NOT NULL,
    key TEXT NOT NULL CHECK (char_length(key) BETWEEN 8 AND 255),
    request_hash TEXT NOT NULL CHECK (request_hash ~ '^[0-9a-f]{64}$'),
    response_status SMALLINT NOT NULL CHECK (response_status BETWEEN 200 AND 599),
    response_body JSONB NOT NULL,
    resource_type TEXT NULL,
    resource_id UUID NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    expires_at TIMESTAMPTZ NOT NULL CHECK (expires_at > created_at),
    -- At-most-once execution per principal, operation, and key
    CONSTRAINT idempotency_keys_principal_op_key_unique UNIQUE (principal_user_id, operation, key)
);

CREATE OR REPLACE FUNCTION fn_trips_immutability()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.passenger_id <> NEW.passenger_id THEN
        RAISE EXCEPTION 'trips.passenger_id is immutable after insert' USING ERRCODE = 'check_violation';
    END IF;
    IF OLD.fare_minor <> NEW.fare_minor THEN
        RAISE EXCEPTION 'trips.fare_minor is immutable after insert' USING ERRCODE = 'check_violation';
    END IF;
    IF OLD.currency <> NEW.currency THEN
        RAISE EXCEPTION 'trips.currency is immutable after insert' USING ERRCODE = 'check_violation';
    END IF;
    IF OLD.payment_method <> NEW.payment_method THEN
        RAISE EXCEPTION 'trips.payment_method is immutable after insert' USING ERRCODE = 'check_violation';
    END IF;
    IF OLD.requested_at <> NEW.requested_at THEN
        RAISE EXCEPTION 'trips.requested_at is immutable after insert' USING ERRCODE = 'check_violation';
    END IF;

    -- Assignment immutability (BR16): driver and vehicle can move from NULL to non-NULL only once
    IF OLD.driver_id IS NOT NULL AND (OLD.driver_id <> NEW.driver_id OR NEW.driver_id IS NULL) THEN
        RAISE EXCEPTION 'trips.driver_id cannot be reassigned once set' USING ERRCODE = 'check_violation';
    END IF;
    IF OLD.vehicle_id IS NOT NULL AND (OLD.vehicle_id <> NEW.vehicle_id OR NEW.vehicle_id IS NULL) THEN
        RAISE EXCEPTION 'trips.vehicle_id cannot be reassigned once set' USING ERRCODE = 'check_violation';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
CREATE FUNCTION
CREATE TRIGGER trg_trips_immutability
    BEFORE UPDATE ON trips
    FOR EACH ROW EXECUTE FUNCTION fn_trips_immutability();

CREATE OR REPLACE FUNCTION fn_trips_state_machine()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.status = NEW.status THEN
        RETURN NEW;
    END IF;

    -- Terminal states: completed, cancelled, expired cannot transition to any other status
    IF OLD.status IN ('completed', 'cancelled', 'expired') THEN
        RAISE EXCEPTION 'Illegal state transition: trip in terminal status % cannot be modified to %', OLD.status, NEW.status
            USING ERRCODE = 'check_violation';
    END IF;

    -- Transition allowlist according to docs/state-machines.md
    IF OLD.status = 'requested' AND NEW.status IN ('accepted', 'cancelled', 'expired') THEN
        RETURN NEW;
    ELSIF OLD.status = 'accepted' AND NEW.status IN ('driver_arriving', 'cancelled') THEN
        RETURN NEW;
    ELSIF OLD.status = 'driver_arriving' AND NEW.status IN ('in_progress', 'cancelled') THEN
        RETURN NEW;
    ELSIF OLD.status = 'in_progress' AND NEW.status = 'completed' THEN
        RETURN NEW;
    ELSE
        RAISE EXCEPTION 'Illegal state transition: cannot transition trip from % to %', OLD.status, NEW.status
            USING ERRCODE = 'check_violation';
    END IF;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_trips_state_machine
    BEFORE UPDATE OF status ON trips
    FOR EACH ROW EXECUTE FUNCTION fn_trips_state_machine();

CREATE OR REPLACE FUNCTION fn_payments_transition()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.status = NEW.status THEN
        -- If settled, cannot update any other field either
        IF OLD.status IN ('succeeded', 'failed') THEN
            RAISE EXCEPTION 'Settled payments are immutable (attempt % for trip %)', OLD.attempt_no, OLD.trip_id
                USING ERRCODE = 'check_violation';
        END IF;
        RETURN NEW;
    END IF;

    IF OLD.status = 'pending' AND NEW.status IN ('succeeded', 'failed') THEN
        RETURN NEW;
    ELSE
        RAISE EXCEPTION 'Illegal payment status transition from % to % (attempt %)', OLD.status, NEW.status, OLD.attempt_no
            USING ERRCODE = 'check_violation';
    END IF;
END;
$$ LANGUAGE plpgsql;

    BEFORE UPDATE ON payments
    FOR EACH ROW EXECUTE FUNCTION fn_payments_transition();
